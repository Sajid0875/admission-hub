tenantScope.middleware.js
function scopeToPartner(req, res, next):
if (req.user.role !== 'super_admin') req.query.partnerId = req.user.partnerId;
next();
— injected on every route touching Lead, Admission, CommissionRecord, Notification

rbac.middleware.js
function requirePermission(permission):
return (req, res, next) => {
if (!req.user.role.permissions.includes(permission))
return res.status(403).json({success:false, error:{code:'FORBIDDEN', ...}});
next();
}

LLD — API Contracts

LLD-API

POST /api/v1/leads
// Request body (Zod-validated)
{
"studentName": "Aarav Sharma", "phone": "+91XXXXXXXXXX", "email": "aarav@example.com",
"courseId": "665f...", "source": "website", "priority": "high", "budget": 45000
}
// Response 201
{ "success": true, "data": { "_id": "66a1...", "partnerId": "6512...", "status": "new",
"leadScore": 62, "isDuplicate": false, "createdAt": "2026-08-28T10:12:00Z" } }
// Response 409 — duplicate detected
{ "success": false, "error": { "code": "DUPLICATE_LEAD",
"message": "A lead with this phone number already exists for this partner" } }

PATCH /api/v1/commissions/{id}/payout
// Request — Super Admin only, requires permission "commission:approve_payout"
{ "action": "approve" }
// Response 200
{ "success": true, "data": { "_id": "...", "payoutStatus": "approved", "payoutDate": "2026-08-28T10:20:00Z" } }
// Response 403 — role lacks permission
{ "success": false, "error": { "code": "FORBIDDEN", "message": "role 'partner_admin' cannot approve payouts" } }

GET /api/v1/leads — list (paginated, tenant-scoped)
// Response 200
{ "success": true,
"data": [ { "_id": "...", "studentName": "...", "status": "follow_up", ... } ],
"meta": { "page": 1, "limit": 20, "total": 134 } }
// tenantScope.middleware.js has already injected partnerId into the Mongoose filter
// before the controller runs — Partner Admin/Team Member can never see another tenant's leads

LLD — Sequence Flow: Lead Create → RBAC → Tenant Scope
1

Client → API — POST /api/v1/leads, header Authorization: Bearer <jwt>

2

auth.middleware.js — verifies JWT signature/expiry, attaches req.user = {id, role, partnerId, permissions}; 401 if invalid

3

rbac.middleware.js — requirePermission('lead:create') checks req.user.role.permissions; 403 if missing

4

tenantScope.middleware.js — injects req.body.partnerId = req.user.partnerId (skipped for Super Admin, who may specify any partnerId)

5

Zod validator — validates studentName, phone format, courseId shape; 400 on failure with field errors

6

LeadController.create() — calls LeadService.create(validatedBody)

7

LeadService — runs duplicateCheck({partnerId, phone}); if found, sets isDuplicate=true or returns 409 per business rule; computes leadScore

8

Mongoose — Lead.create(doc) — persisted to MongoDB Atlas

9

NotificationService — fires "lead assigned" notification to the owner (async, does not block the response)

LLD-SEQ

10 Controller → Client — 201, { success: true, data: {...} }

Two independent gates run before any business logic executes: rbac.middleware.js ("is this role allowed to create leads at all?") and tenantScope.middleware.js ("which
tenant does this write belong to?"). This mirrors the SRS's core rule — a partner can access only its own tenant's data — enforced at the data-access layer, not the UI.
